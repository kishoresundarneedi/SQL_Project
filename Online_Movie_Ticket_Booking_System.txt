================================
PART 1: DATABASE & TABLE CREATION
================================
CREATE DATABASE online_movie_ticket_booking;
USE online_movie_ticket_booking;

--CUSTOMERS
CREATE TABLE customers (
    customer_id INT AUTO_INCREMENT PRIMARY KEY,
    customer_name VARCHAR(100),
    phone VARCHAR(15),
    email VARCHAR(100)
);

-- THEATRES
CREATE TABLE theatres (
    theatre_id INT AUTO_INCREMENT PRIMARY KEY,
    theatre_name VARCHAR(100),
    city VARCHAR(50),
    total_screens INT
);

-- MOVIES
CREATE TABLE movies (
    movie_id INT AUTO_INCREMENT PRIMARY KEY,
    movie_name VARCHAR(100),
    genre VARCHAR(50),
    language VARCHAR(50),
    duration_minutes INT,
    release_date DATE
);

-- SHOWS
CREATE TABLE shows (
    show_id INT AUTO_INCREMENT PRIMARY KEY,
    movie_id INT,
    theatre_id INT,
    show_date DATE,
    show_time TIME,
    ticket_price DECIMAL(10,2),
    available_seats INT,
    FOREIGN KEY(movie_id) REFERENCES movies(movie_id),
    FOREIGN KEY(theatre_id) REFERENCES theatres(theatre_id)
);

-- BOOKINGS
CREATE TABLE bookings (
    booking_id INT AUTO_INCREMENT PRIMARY KEY,
    customer_id INT,
    booking_date DATE,
    total_amount DECIMAL(10,2),
    payment_mode ENUM('UPI','Cash','Card'),
    FOREIGN KEY(customer_id)
    REFERENCES customers(customer_id)
);

-- BOOKING DETAILS (Weak Entity)
CREATE TABLE booking_details (
    ticket_id INT AUTO_INCREMENT PRIMARY KEY,
    booking_id INT,
    show_id INT,
    seats_booked INT,
    subtotal DECIMAL(10,2),
    FOREIGN KEY(booking_id)
    REFERENCES bookings(booking_id),
    FOREIGN KEY(show_id)
    REFERENCES shows(show_id)
);

PART  4: VIEWS
=========================
CREATE VIEW view_booking_summary AS
SELECT
b.booking_id,
c.customer_name,
b.booking_date,
b.total_amount,
b.payment_mode
FROM bookings b
JOIN customers c
ON b.customer_id=c.customer_id;

CREATE VIEW view_low_seats AS
SELECT *
FROM shows
WHERE available_seats < 100;

CREATE VIEW view_customer_spend AS
SELECT
c.customer_id,
c.customer_name,
COUNT(b.booking_id) total_bookings,
SUM(b.total_amount) total_spent
FROM customers c
LEFT JOIN bookings b
ON c.customer_id=b.customer_id
GROUP BY c.customer_id;

PART  5: STORED PROCEDURES

DELIMITER $$

CREATE PROCEDURE sp_customer_bookings(
IN p_customer_id INT
)
BEGIN
SELECT *
FROM bookings
WHERE customer_id=p_customer_id;
END $$

DELIMITER ;


DELIMITER $$

CREATE PROCEDURE sp_book_ticket(
IN p_customer_id INT,
IN p_show_id INT,
IN p_seats INT,
IN p_payment_mode VARCHAR(10)
)

BEGIN

DECLARE v_price DECIMAL(10,2);
DECLARE v_total DECIMAL(10,2);

SELECT ticket_price
INTO v_price
FROM shows
WHERE show_id=p_show_id;

SET v_total=v_price*p_seats;

INSERT INTO bookings
(customer_id,booking_date,total_amount,payment_mode)
VALUES
(p_customer_id,CURDATE(),v_total,p_payment_mode);

END $$

DELIMITER ;


DELIMITER $$

CREATE PROCEDURE sp_revenue_report()
BEGIN

SELECT
payment_mode,
SUM(total_amount) total_revenue
FROM bookings
GROUP BY payment_mode;

END $$

PART  6: TRIGGERS
========================
DELIMITER $$

CREATE TRIGGER trg_reduce_seats

AFTER INSERT ON booking_details

FOR EACH ROW

BEGIN

UPDATE shows
SET available_seats =
available_seats - NEW.seats_booked
WHERE show_id=NEW.show_id;

END $$

DELIMITER ;
DELIMITER $$

CREATE TRIGGER trg_prevent_overbooking

BEFORE INSERT ON booking_details

FOR EACH ROW

BEGIN

DECLARE v_seats INT;

SELECT available_seats
INTO v_seats
FROM shows
WHERE show_id=NEW.show_id;

IF v_seats < NEW.seats_booked THEN

SIGNAL SQLSTATE '45000'
SET MESSAGE_TEXT='Seats Not Available';

END IF;

END $$

DELIMITER ;

Q10 Most Popular Movie
SQL
SELECT
m.movie_name,
SUM(bd.seats_booked) tickets
FROM movies m
JOIN shows s
ON m.movie_id=s.movie_id
JOIN booking_details bd
ON s.show_id=bd.show_id
GROUP BY m.movie_name
ORDER BY tickets DESC
LIMIT 1;
